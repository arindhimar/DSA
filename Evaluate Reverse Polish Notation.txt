class Solution {
public:
    int evalRPN(vector<string>& tokens) {
        stack<int> st;

        for (string& s : tokens) {
            if (s == "+" || s == "-" || s == "*" || s == "/") {
                int n2 = st.top(); st.pop();  
                int n1 = st.top(); st.pop();  
                int res = 0;

                if (s == "+") res = n1 + n2;
                else if (s == "-") res = n1 - n2;
                else if (s == "*") res = n1 * n2;
                else if (s == "/") res = n1 / n2;

                st.push(res);
            } else {
                st.push(stoi(s));
            }
        }

        return st.top();
    }
};
